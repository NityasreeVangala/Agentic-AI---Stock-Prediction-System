```python
# backend/app/services/fundamentals.py
from app.services.snapshot import generate_snapshot

# Sample normalized fundamentals (Day 12)
normalized_fundamentals = {
    "TCS.NS": {"PE": 0.866, "EPS": 1, "ROE": 0.7, "Debt": 0.09},
    "RELIANCE.NS": {"PE": 0.666, "EPS": 0.7, "ROE": 1, "Debt": 0.27},
    "HDFCBANK.NS": {"PE": 0.466, "EPS": 0.8, "ROE": 0.6, "Debt": 0.18},
    "YESBANK.NS": {"PE": 0, "EPS": 0, "ROE": 0, "Debt": 1},
    "INFY.NS": {"PE": 1, "EPS": 0.9, "ROE": 0.8, "Debt": 0},
}

# Layman explanations (Day 14)
metric_explanations = {
    "PE": "PE is high, stock is slightly expensive",
    "EPS": "Earnings per share are high",
    "ROE": "Returns are decent",
    "Debt": "Low debt, financially safe"
}

def calculate_score(stock):
    """
    Calculate 0-100 score using normalized fundamentals
    Weights: PE 20%, EPS 30%, ROE 30%, Debt 20%
    """
    f = normalized_fundamentals[stock]
    score = (f["PE"]*0.2 + f["EPS"]*0.3 + f["ROE"]*0.3 + (1-f["Debt"])*0.2) * 100
    return round(score, 2)

def get_scores():
    """Return scores for all stocks"""
    return {stock: calculate_score(stock) for stock in normalized_fundamentals.keys()}

def generate_positives_risks():
    """
    Generate top positives and risks for each stock
    - Positives: metrics with high value
    - Risks: metrics with low value or high debt
    """
    result = {}
    for stock, f in normalized_fundamentals.items():
        positives = []
        risks = []
        for metric, val in f.items():
            if metric != "Debt":
                if val >= 0.7:
                    positives.append(metric_explanations[metric])
                else:
                    risks.append(metric_explanations[metric])
            else:
                if val <= 0.3:
                    positives.append(metric_explanations[metric])
                else:
                    risks.append(metric_explanations[metric])
        result[stock] = {"positives": positives, "risks": risks}
    return result
```

```python
from app.data.fundamentals_data import fundamentals

def rank_metrics():
    metrics = ["PE", "EPS", "ROE", "Debt"]
    stocks = list(fundamentals.keys())
    ranked = {s: {} for s in stocks}

    for metric in metrics:
        reverse = metric != "Debt"
        sorted_stocks = sorted(
            stocks,
            key=lambda s: fundamentals[s][metric],
            reverse=reverse
        )
        for rank, stock in enumerate(sorted_stocks, start=1):
            ranked[stock][metric] = rank

    return ranked


def layman_insights(stock):
    f = fundamentals[stock]
    return {
        "PE": "Expensive" if f["PE"] > 0.7 else "Cheap",
        "EPS": "Strong earnings" if f["EPS"] > 0.7 else "Weak earnings",
        "ROE": "High returns" if f["ROE"] > 0.7 else "Low returns",
        "Debt": "Low debt" if f["Debt"] < 0.3 else "High debt"
    }


def top5_positives_risks(stock):
    f = fundamentals[stock]
    positives, risks = [], []

    if f["PE"] < 0.3: positives.append("Stock is undervalued")
    if f["EPS"] > 0.7: positives.append("Strong earnings growth")
    if f["ROE"] > 0.7: positives.append("Excellent profitability")
    if f["Debt"] < 0.3: positives.append("Low debt burden")

    if f["PE"] > 0.7: risks.append("Stock may be overpriced")
    if f["EPS"] < 0.3: risks.append("Weak earnings")
    if f["ROE"] < 0.3: risks.append("Poor profitability")
    if f["Debt"] > 0.7: risks.append("High debt risk")

    return {
        "positives": positives[:5],
        "risks": risks[:5]
    }
```
