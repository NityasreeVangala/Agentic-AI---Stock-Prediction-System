```python
import pandas as pd
import xgboost as xgb
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, mean_absolute_error
import joblib  # to save model

def train_xgb(stock_csv_path="app/data/stock_history.csv"):
    """
    Train XGBoost regression to predict next-day closing price.
    """
    df = pd.read_csv(stock_csv_path, parse_dates=["Date"])
    df = df.sort_values("Date")
    
    # --- Feature engineering ---
    # Lag features (previous 1,2,3 days)
    for lag in range(1, 4):
        df[f"Close_lag{lag}"] = df["Close"].shift(lag)

    # Rolling mean
    df["MA3"] = df["Close"].rolling(3).mean()
    df["MA7"] = df["Close"].rolling(7).mean()

    # Drop rows with NaN due to lags/rolling
    df = df.dropna()

    # Features & target
    feature_cols = ["Close_lag1", "Close_lag2", "Close_lag3", "MA3", "MA7"]
    X = df[feature_cols]
    y = df["Close"]

    # Train/test split
    X_train, X_test, y_train, y_test = train_test_split(X, y, shuffle=False, test_size=0.2)

    # Train model
    model = xgb.XGBRegressor(n_estimators=200, max_depth=5, learning_rate=0.1)
    model.fit(X_train, y_train)

    # Predictions
    y_pred = model.predict(X_test)

    # Metrics
    rmse = mean_squared_error(y_test, y_pred, squared=False)
    mae = mean_absolute_error(y_test, y_pred)
    print(f"XGBoost RMSE: {rmse:.2f}, MAE: {mae:.2f}")

    # Save model
    joblib.dump(model, "app/services/xgb_model.pkl")
    print("XGBoost model saved!")

    return model, X_test, y_test, y_pred
```

* Saves model as **`xgb_model.pkl`** for future prediction API.

# **2️⃣ LSTM Sequence Modeling (Day 46–50)**

**File:** `backend/app/services/forecasting_lstm.py`

```python
import pandas as pd
import numpy as np
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense
from sklearn.preprocessing import MinMaxScaler
import joblib

def train_lstm(stock_csv_path="app/data/stock_history.csv", timesteps=30):
    """
    Train LSTM to predict next-day closing price.
    """
    df = pd.read_csv(stock_csv_path, parse_dates=["Date"])
    df = df.sort_values("Date")

    # Only use 'Close' for LSTM
    close_prices = df["Close"].values.reshape(-1, 1)

    # Normalize
    scaler = MinMaxScaler()
    close_scaled = scaler.fit_transform(close_prices)

    # Create sequences
    X, y = [], []
    for i in range(timesteps, len(close_scaled)):
        X.append(close_scaled[i-timesteps:i, 0])
        y.append(close_scaled[i, 0])
    X, y = np.array(X), np.array(y)

    # Reshape for LSTM: [samples, timesteps, features]
    X = X.reshape(X.shape[0], X.shape[1], 1)

    # Split 80/20
    split = int(0.8 * len(X))
    X_train, X_test = X[:split], X[split:]
    y_train, y_test = y[:split], y[split:]

    # Build model
    model = Sequential()
    model.add(LSTM(50, input_shape=(X_train.shape[1], 1)))
    model.add(Dense(1))
    model.compile(optimizer="adam", loss="mse")

    # Train
    model.fit(X_train, y_train, epochs=50, batch_size=32, verbose=1)

    # Save model + scaler
    model.save("app/services/lstm_model.h5")
    joblib.dump(scaler, "app/services/lstm_scaler.pkl")
    print("LSTM model and scaler saved!")

    return model, X_test, y_test
```

✅ **Notes:**

* Input sequence length (`timesteps`) = 30 days.
* Scales data to [0,1] for neural network.
* Saves model as **`lstm_model.h5`** and scaler as **`lstm_scaler.pkl`** for later inference.

## ✅ **Day 41–45: XGBoost Regression**

**Objective:** Train a tree-based model (XGBoost) using **lag features** and **rolling averages**.

### 1️⃣ Features

We use:

1. **Lag features:** Previous 1, 2, 3 days of closing prices
   `Close_lag1, Close_lag2, Close_lag3`
2. **Rolling averages:** 3-day and 7-day moving averages
   `MA3, MA7`

### 2️⃣ Code: `forecasting_xgb.py`

```python
import pandas as pd
import xgboost as xgb
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, mean_absolute_error
import joblib  # Save model

def train_xgb(stock_csv_path="app/data/all_stock_prices.csv"):
    """
    Train XGBoost regression to predict next-day closing price.
    """
    print(f"✅ Loading CSV: {stock_csv_path}")
    df = pd.read_csv(stock_csv_path, parse_dates=["Date"])
    df = df.sort_values("Date")
    
    # Lag features
    for lag in range(1, 4):
        df[f"Close_lag{lag}"] = df["Close"].shift(lag)

    # Rolling averages
    df["MA3"] = df["Close"].rolling(3).mean()
    df["MA7"] = df["Close"].rolling(7).mean()

    # Drop NaN
    df = df.dropna()

    # Features & target
    feature_cols = ["Close_lag1", "Close_lag2", "Close_lag3", "MA3", "MA7"]
    X = df[feature_cols]
    y = df["Close"]

    # Train/test split (80%/20%)
    split_index = int(0.8 * len(X))
    X_train, X_test = X[:split_index], X[split_index:]
    y_train, y_test = y[:split_index], y[split_index:]

    # Train XGBoost
    model = xgb.XGBRegressor(n_estimators=200, max_depth=5, learning_rate=0.1)
    model.fit(X_train, y_train)
    print("✅ XGBoost model trained!")

    # Predictions
    y_pred = model.predict(X_test)

    # Metrics
    rmse = mean_squared_error(y_test, y_pred)**0.5
    mae = mean_absolute_error(y_test, y_pred)
    print(f"XGBoost RMSE: {rmse:.2f}, MAE: {mae:.2f}")

    # Save model
    joblib.dump(model, "app/services/models/xgb_model.pkl")
    print("✅ XGBoost model saved!")

    return model, X_test, y_test, y_pred

if __name__ == "__main__":
```

## ✅ **Day 46–50: LSTM Model**

**Objective:** Use **sequence modeling** to capture trends and patterns in stock prices.

* **LSTM (Long Short-Term Memory)** is a type of recurrent neural network for time series.
* Input: sequences of `timesteps` days of **closing prices**.
* Output: next-day closing price.

### 1️⃣ Features

* Only **Close price**.
* Normalize prices using **MinMaxScaler** (0–1 scale).
* Create **sliding windows** of `timesteps=30` (previous 30 days).

### 2️⃣ Code: `forecasting_lstm.py`

```python
import pandas as pd
import numpy as np
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense
from sklearn.preprocessing import MinMaxScaler
import joblib

def train_lstm(stock_csv_path="app/data/all_stock_prices.csv", timesteps=30):
    """
    Train LSTM to predict next-day closing price.
    """
    df = pd.read_csv(stock_csv_path, parse_dates=["Date"])
    df = df.sort_values("Date")

    # Only Close prices
    close_prices = df["Close"].values.reshape(-1, 1)

    # Normalize
    scaler = MinMaxScaler()
    close_scaled = scaler.fit_transform(close_prices)

    # Create sequences
    X, y = [], []
    for i in range(timesteps, len(close_scaled)):
        X.append(close_scaled[i-timesteps:i, 0])
        y.append(close_scaled[i, 0])
    X, y = np.array(X), np.array(y)
    X = X.reshape(X.shape[0], X.shape[1], 1)  # [samples, timesteps, features]

    # Train/test split
    split = int(0.8 * len(X))
    X_train, X_test = X[:split], X[split:]
    y_train, y_test = y[:split], y[split:]

    # Build LSTM
    model = Sequential()
    model.add(LSTM(50, input_shape=(X_train.shape[1], 1)))
    model.add(Dense(1))
    model.compile(optimizer="adam", loss="mse")

    # Train
    model.fit(X_train, y_train, epochs=50, batch_size=32, verbose=1)

    # Save model and scaler
    model.save("app/services/lstm_model.h5")
    joblib.dump(scaler, "app/services/models/lstm_scaler.pkl")
    print("✅ LSTM model and scaler saved!")

    return model, X_test, y_test

if __name__ == "__main__":
    train_lstm()
```
